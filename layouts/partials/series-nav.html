{{- /* Series navigation bar — "Part X of N · ← Prev | Next →". Renders only when the page belongs to a series with more than one post. Order is chronological (ByDate), which matches the numbered part order. */ -}} {{- with .GetTerms "series" -}} {{- $cur := $ -}} {{- $term := index . 0 -}} {{- $pages := $term.Pages.ByDate -}} {{- if $cur.Param "seriesWeight" -}} {{- $pages = $term.Pages.ByParam "seriesWeight" -}} {{- end -}} {{- $total := len $pages -}} {{- if gt $total 1 -}} {{- $idx := -1 -}} {{- range $i, $p := $pages -}} {{- if eq $p.RelPermalink $cur.RelPermalink -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- if ge $idx 0 -}} {{- $prev := false -}} {{- if gt $idx 0 -}}{{- $prev = index $pages (sub $idx 1) -}}{{- end -}} {{- $next := false -}} {{- if lt $idx (sub $total 1) -}}{{- $next = index $pages (add $idx 1) -}}{{- end -}}
{{ $term.LinkTitle }} Part {{ add $idx 1 }} of {{ $total }}
{{- with $prev }} ← Previous {{ .LinkTitle }} {{- else }}{{ end }} {{- with $next }} Next → {{ .LinkTitle }} {{- else }}{{ end }}
{{- end -}} {{- end -}} {{- end -}}
